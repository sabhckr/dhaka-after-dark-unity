# builds Dad.World, the survey and the test mirror (P6Check.cs) in the Mono harness, then runs the survey on the slice
# against the known list and on Line 6 (64 chunks). Caught = the slice gate fails, Line 6 shows an island or median
# finding on a viaduct pier or other figures than today's (23 piers on an island, 331 on a carriageway without), or
# a mirrored test fails.

only = sys.argv[2].split(',') if len(sys.argv) > 2 and sys.argv[2] else None
L6 = open(H + '/../line6-ids.txt').read().strip()
def survey(d, tag, ids=None):
    o = f'{d}/{tag}'
    os.makedirs(o, exist_ok=True)
    subprocess.run([MO + '/bin/mono', f'{d}/infra.exe', CH, o, P + '/Tests/Infra/known-violations.csv'] + ([ids] if ids else []), env=ENV, check=True, capture_output=True)
    return o
for name, rel, a, b in M:
    tag = name.split()[0]
    if only and tag not in only: continue
    d = f'{OUT}/{tag}'
    if os.path.exists(d): shutil.rmtree(d)
    os.makedirs(d + '/World'); os.makedirs(d + '/Infra')
    for f in glob.glob(P + '/Assets/_Project/Runtime/World/*.cs'): shutil.copy(f, d + '/World')
    for f in glob.glob(P + '/Assets/_Project/Editor/Infra/*.cs'): shutil.copy(f, d + '/Infra')
    src = open(f'{d}/{rel}').read()
    assert src.count(a) == 1, (name, src.count(a))
    open(f'{d}/{rel}', 'w').write(src.replace(a, b))
    subprocess.run(CSC + ['-target:library', '-unsafe', f'-out:{d}/Dad.World.dll', '-r:' + NS] + glob.glob(d + '/World/*.cs'), check=True, capture_output=True)
    subprocess.run(CSC + ['-target:exe', f'-out:{d}/infra.exe', '-r:' + NS, f'-r:{d}/Dad.World.dll', '-r:' + MS] + glob.glob(d + '/Infra/*.cs') + [P + '/Tools/infra/Infra.cs'], check=True, capture_output=True)
    subprocess.run(CSC + ['-target:exe', f'-out:{d}/p6check.exe', '-r:' + NS, f'-r:{d}/Dad.World.dll', '-r:' + MS, H + '/P6Check.cs'], check=True, capture_output=True)
    why = []
    sl = survey(d, 'slice')
    summ = open(sl + '/summary.txt').read()
    gate = [l for l in summ.splitlines() if l.startswith('Against')][0]
    if ' 0 new, 0 gone' not in gate:
        ex = [l.strip() for l in summ.splitlines() if l.strip().startswith(('NEW', 'GONE'))][:1]
        why.append('slice gate ' + gate.split(': ')[1] + (' e.g. ' + ex[0][:150] if ex else ''))
    l6 = survey(d, 'line6', L6)
    isl = left = 0; checks = collections.Counter()
    for r in csv.DictReader(open(l6 + '/supports.csv')):
        if not r['id'].startswith('VP-') or r['built'] != 'yes': continue
        if r['builder_note'] == 'median': isl += 1
        elif r['builder_note'].startswith('on a carriageway'): left += 1
        for f in r['findings'].split(' | '):
            c = f.split(':')[0]
            if c.startswith(('median', 'island')): checks[c] += 1
    if checks: why.append('Line 6 findings ' + dict(checks).__repr__())
    if (isl, left) != (23, 331): why.append(f'Line 6 figures {isl} on an island, {left} left (pinned 23, 331)')
    m = subprocess.run([MO + '/bin/mono', f'{d}/p6check.exe'], env=ENV, capture_output=True, text=True).stdout.strip()
    if m != 'OK': why.append('mirrored tests: ' + m)
    line = ('CAUGHT  ' if why else 'MISSED  ') + name + ': ' + ('; '.join(why) if why else 'nothing failed')
    print(line, flush=True)
    shutil.rmtree(d + '/World'); shutil.rmtree(d + '/Infra')
