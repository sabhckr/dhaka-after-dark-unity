# Infrastructure checker mutations: each breaks one check (survey) or one placement rule (builder), runs the
# survey against the known list in the Mono harness (Tools/infra/run.sh) and restores the source.
import subprocess, sys, os, re
P = os.path.expanduser('~/Documents/dhaka-after-dark-unity')
S = P + '/Assets/_Project/Editor/Infra/InfraSurvey.cs'
B = P + '/Assets/_Project/Runtime/World/Structures.cs'
M = [
 ('S1 survey counts footways as carriageways', S, 'if (foot.Contains(l.Way)) continue;', 'if (false && foot.Contains(l.Way)) continue;'),
 ('S2 survey duplicate limit back to 5 m', S, 'DuplicatePier = ChunkRoads.TwinMax,', 'DuplicatePier = 5,'),
 ('S3 survey spans ignore which deck', S, '|| !s.Tracks.Overlaps(o.Tracks)) continue;', ') continue;'),
 ('S4 survey drops the raised-building check', S, 'else if (s.BuildingBase < s.Top) Add("raised"', 'else if (false) Add("raised"'),
 ('S5 survey joins no decks at their ends', S, 'JoinReach = 0.5;', 'JoinReach = -10;'),
 ('S6 survey deck-top tolerance broken', S, 'Math.Abs(s.Top - s.ExpectedTop) > TopTolerance', 'Math.Abs(s.Top - s.ExpectedTop + 0.3) > TopTolerance'),
 ('B1 builder: footbridge columns ignore carriageways', B, '!InChunk(x, z) || OnCarriageway(x, z, 0.28, 0.3)', '!InChunk(x, z)'),
 ('B2 builder: piers no longer skip junctions', B, 'inPad = InAnyPad(x, z, 0.28, PierRadius + 0.5);', 'inPad = false;'),
 ('B3 builder: pier grid 25 m', B, 'public const double PierGrid = 30,', 'public const double PierGrid = 25,'),
 ('B4 builder: stairs reach 1 m', B, 'StairReach = 2.5;', 'StairReach = 1;'),
]
out = sys.argv[1]
for name, f, a, b in M:
    src = open(f).read()
    assert src.count(a) == 1, (name, src.count(a))
    open(f, 'w').write(src.replace(a, b))
    try:
        r = subprocess.run([P + '/Tools/infra/run.sh', out + '/' + name.split()[0]], capture_output=True, text=True)
        txt = r.stdout + r.stderr
        m = re.search(r'Against the known list \((\d+) entries\): (\d+) new, (\d+) gone', txt)
        res = f'{m.group(2)} new, {m.group(3)} gone' if m else 'NO RESULT: ' + txt[-300:]
        caught = m and (m.group(2) != '0' or m.group(3) != '0')
        ex = [l.strip() for l in txt.splitlines() if l.strip().startswith(('NEW ', 'GONE '))][:2]
        print(f"{'CAUGHT' if caught else 'MISSED'}  {name}: {res}  e.g. {' / '.join(x[:140] for x in ex)}")
    finally:
        open(f, 'w').write(src)
