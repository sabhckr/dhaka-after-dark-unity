for name, f, a, b, mode in muts:
    path=os.path.join(P,f); bak=os.path.join(D,'backup',os.path.basename(f))
    shutil.copy2(path,bak)
    s=open(path).read()
    if s.count(a)!=1: out.write(f'## {name}: NOT APPLIED (matches {s.count(a)})\n'); out.flush(); continue
    open(path,'w').write(s.replace(a,b))
    try:
        if mode=='Mono':
            env=dict(os.environ, EXTRA=f'{MT}/D06c.cs {MT}/ShimD06c.cs', TESTS=f'{MT}/NearFacadeTestsMono.cs')
            r=subprocess.run(['/bin/zsh',f'{MT}/run.sh']+MONO,cwd=MT,capture_output=True,text=True,env=env)
            lines=[l for l in (r.stdout+r.stderr).splitlines() if l.startswith(('PASS','FAIL','/'))]
            caught=sum(1 for l in lines if l.startswith('FAIL')) or (1 if r.returncode and not any(l.startswith('PASS') for l in lines) else 0)
            out.write(f'## {name} [{mode}]: {"CAUGHT" if caught else "MISSED"} (exit {r.returncode})\n'+'\n'.join(l[:220] for l in lines if not l.startswith('PASS'))+'\n'); out.flush()
        else:
            r=subprocess.run(['caffeinate','-dimsu','Tools/run-tests.sh','PlayMode','Dad.Tests.ChunkStreamerTests'],cwd=P,capture_output=True,text=True)
            tail=r.stdout.strip().splitlines()[-8:]
            out.write(f'## {name} [{mode}]: {"CAUGHT" if r.returncode else "MISSED"} (exit {r.returncode})\n'+'\n'.join(tail)+'\n'); out.flush()
    finally:
        shutil.copy2(bak,path)
out.write('restored\n'); out.close()
