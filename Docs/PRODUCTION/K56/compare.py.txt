# K-56: compare profiler markers of two players (2026-11-05).
#   python3 -I compare.py.txt <thread> main=<markers.tsv>[,<markers.tsv>...] p5a=<markers.tsv>[,...] [top]
# thread: "Main Thread" or "Render Thread". For every marker name: mean self and total ms per frame in each capture (a name met
# at several places in the hierarchy is summed; rows under 0.05 ms self and total are not in the TSV, see CaptureReport.cs).
# Prints the markers whose mean differs most between the two players (difference of the players' means), with every
# capture's own value, so a difference that one capture alone makes can be seen. Also "busy": PlayerLoop total minus the
# markers that wait (name contains "Wait"), as the bench's main-thread time counts it.
import sys, collections
thread = sys.argv[1]; groups = {}; top = 25
for a in sys.argv[2:]:
    if '=' in a: k, v = a.split('=', 1); groups[k] = v.split(',')
    else: top = int(a)
def load(p):
    lines = open(p).read().split('\n'); r = lines[0].split('frames ')[1].split('..'); n = int(r[1]) - int(r[0]) + 1
    s = collections.Counter(); t = collections.Counter(); c = collections.Counter(); wait = 0.0; loop = 0.0
    for l in lines[2:]:
        f = l.split('\t')
        if len(f) < 7 or f[1] != thread: continue
        s[f[3]] += float(f[4]); t[f[3]] += float(f[5]); c[f[3]] += float(f[6])
    for k in t:
        if 'Wait' in k: wait += s[k]
    return n, {k: (s[k] / n, t[k] / n, c[k] / n) for k in t}, (t.get('PlayerLoop', 0) - wait) / n
data = {g: [load(p) for p in ps] for g, ps in groups.items()}
(ga, A), (gb, B) = list(data.items())
names = set(k for _, d, _ in A + B for k in d)
mean = lambda caps, k, i: sum(d.get(k, (0, 0, 0))[i] for _, d, _ in caps) / len(caps)
print('frames:', {g: [n for n, _, _ in v] for g, v in data.items()})
print('busy (PlayerLoop minus waits) ms/frame:', {g: ['%.3f' % b for _, _, b in v] for g, v in data.items()})
for i, label in ((1, 'total'), (0, 'self')):
    print('\n== %s ms per frame: %s - %s, largest differences' % (label, gb, ga))
    rows = sorted(names, key=lambda k: -abs(mean(B, k, i) - mean(A, k, i)))[:top]
    for k in rows:
        print('%+.3f  %-60s %s %s | %s %s | calls %s / %s' % (mean(B, k, i) - mean(A, k, i), k[:60], ga, ' '.join('%.3f' % d.get(k, (0, 0, 0))[i] for _, d, _ in A),
              gb, ' '.join('%.3f' % d.get(k, (0, 0, 0))[i] for _, d, _ in B), '%.1f' % mean(A, k, 2), '%.1f' % mean(B, k, 2)))
