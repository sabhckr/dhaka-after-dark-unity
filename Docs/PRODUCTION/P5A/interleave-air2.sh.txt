#!/bin/zsh
# P-5a order check (2026-11-05): air path only, MAIN first in each pair, 3 pairs after one discarded run of each. Players as
# interleave.sh.txt: main 88b7a22 (Builds/MacBench-88b7a22) and P-5a (Builds/MacBench). Both are kept.
#   zsh Docs/PRODUCTION/P5A/interleave-air2.sh.txt
D=${0:a:h}; L=$D/interleave-air2.log; cd ~/Documents/dhaka-after-dark-unity
st() { echo "$1 $(date +%H%M%S)" >> $L; }
st "start $(pmset -g batt | head -1) $(ioreg -r -k AppleClamshellState -d 4 | /usr/bin/grep -m1 AppleClamshellState)"
cur=p5a
lid() { ioreg -r -k AppleClamshellState -d 4 | /usr/bin/grep -m1 AppleClamshellState | /usr/bin/grep -q Yes; }
sw() { if [ $cur = main ]; then mv Builds/MacBench Builds/MacBench-88b7a22; mv Builds/MacBench-p5a Builds/MacBench; cur=p5a; else mv Builds/MacBench Builds/MacBench-p5a; mv Builds/MacBench-88b7a22 Builds/MacBench; cur=main; fi; }
stop() { st "lid closed: stopping"; [ $cur = p5a ] || sw; exit 2; }
lid && stop; sw; Tools/bench.sh slice-crossing-air > /dev/null 2>&1; st "main discard exit $? $(ls -td Evidence/local/bench-slice-crossing-air-* | head -1)"
lid && stop; sw; Tools/bench.sh slice-crossing-air > /dev/null 2>&1; st "p5a discard exit $? $(ls -td Evidence/local/bench-slice-crossing-air-* | head -1)"
for i in 1 2 3; do
  for b in main p5a; do
    lid && stop; [ $cur = $b ] || sw
    Tools/bench.sh slice-crossing-air > /dev/null 2>&1; st "$b slice-crossing-air exit $? $(ls -td Evidence/local/bench-slice-crossing-air-* | head -1)"
  done
done
[ $cur = p5a ] || sw
st "done; P-5a player in Builds/MacBench, main in Builds/MacBench-88b7a22 $(pmset -g batt | head -1) $(ioreg -r -k AppleClamshellState -d 4 | /usr/bin/grep -m1 AppleClamshellState)"
