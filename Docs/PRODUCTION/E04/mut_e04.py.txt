caught = survived = 0
for tag, f, tests, a, b in M:
    if TAGS and tag.split()[0] not in TAGS: continue
    src = OUT + '/src'
    shutil.rmtree(src, ignore_errors=True)
    os.makedirs(src)
    for d in ('World', 'Traffic', 'Vehicles'): shutil.copytree(P + '/Assets/_Project/Runtime/' + d, src + '/' + d, ignore=shutil.ignore_patterns('*.meta'))
    s = open(src + '/' + f).read()
    if a not in s: print('NOT FOUND', tag, flush=True); survived += 1; continue
    open(src + '/' + f, 'w').write(s.replace(a, b))
    try:
        r = subprocess.run([RUN, OUT + '/run', tests], env=dict(os.environ, SRC=src), capture_output=True, text=True, timeout=900)
    except subprocess.TimeoutExpired:
        print('CAUGHT   ' + tag + ' <- hung', flush=True); caught += 1; continue
    fails = [l.split(':')[0].split('.')[-1] for l in r.stdout.splitlines() if l.startswith('FAIL')]
    if r.returncode and not fails: print('BUILD    ' + tag + ' ' + (r.stdout + r.stderr)[-300:], flush=True); survived += 1; continue
    if fails: caught += 1
    else: survived += 1
    print(('CAUGHT   ' if fails else 'SURVIVED ') + tag + (' <- ' + ', '.join(fails[:4]) if fails else ''), flush=True)
print(f'{caught} caught, {survived} survived', flush=True)
